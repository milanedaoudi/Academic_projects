===========================================
    Build and Testing Instructions
===========================================
Project: 3A-SEI-SoC-Cpp
Author: Ordan
Objective: Compilation and Valgrind analysis.

-------------------------------------------
# Create the build directory
mkdir -p build
cd build
================== COMPILE ALL =========================
# Generate Makefile files
cmake ..
-------------------------------------------
# Compile all exercises and tests
make
ctest --verbose

# Compile a specific test
make test_name
./test_name 

# Compile MAIN.CPP
make main
./main ../data/platformfile.txt numberSteps -v         #-v for verbose (optional)
================== VALGRIND =========================
# Check for memory leaks and errors
valgrind --leak-check=full ./test_name    (without .cpp)

# Detailed option (shows all allocations and their origins)
valgrind --leak-check=full --show-leak-kinds=all --track-origins=yes ./test_name

------------------ END -------------------------
